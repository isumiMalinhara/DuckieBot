cat > /mnt/user-data/outputs/image_processing_experiments.py << 'EOFPYTHON'
#!/usr/bin/env python3

import cv2
import numpy as np
import matplotlib.pyplot as plt
from pathlib import Path

# Configuration
INPUT_IMAGE = "/mnt/user-data/outputs/lane_frame.jpg"
OUTPUT_DIR = "/mnt/user-data/outputs/experiments"
Path(OUTPUT_DIR).mkdir(exist_ok=True)

def load_image():
    img = cv2.imread(INPUT_IMAGE)
    if img is None:
        raise FileNotFoundError(f"Image not found: {INPUT_IMAGE}")
    return img

def save_image(img, name):
    path = f"{OUTPUT_DIR}/{name}.jpg"
    cv2.imwrite(path, img)
    print(f"Saved: {path}")
    return path

# ============= EXPERIMENT 1: CANNY EDGE DETECTION =============

def experiment_canny(img):
    """Test Canny edge detection with 3 different threshold pairs"""
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    
    # Apply Gaussian blur to reduce noise
    blurred = cv2.GaussianBlur(gray, (5, 5), 0)
    
    # Three threshold pairs
    thresholds = [
        (50, 150, "Low sensitivity - fewer edges"),
        (100, 200, "Medium sensitivity - balanced"),
        (150, 250, "High sensitivity - more edges")
    ]
    
    print("\n=== EXPERIMENT 1: CANNY EDGE DETECTION ===")
    results = []
    
    for i, (low, high, comment) in enumerate(thresholds, 1):
        edges = cv2.Canny(blurred, low, high)
        filename = f"canny_{i}_low{low}_high{high}"
        save_image(edges, filename)
        
        print(f"Canny {i}: Low={low}, High={high}")
        results.append({
            'low': low,
            'high': high,
            'filename': filename,
            'comment': comment
        })
    
    return results

# ============= EXPERIMENT 2: HOUGH TRANSFORM =============

def experiment_hough(img, canny_output):
    """Test Hough Transform with varying threshold parameter"""
    
    # Use Canny edge detection first (medium sensitivity)
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    blurred = cv2.GaussianBlur(gray, (5, 5), 0)
    edges = cv2.Canny(blurred, 100, 200)
    
    # Three threshold values for Hough
    thresholds = [
        (30, "Low threshold - many lines detected"),
        (50, "Medium threshold - balanced detection"),
        (80, "High threshold - only strong lines")
    ]
    
    print("\n=== EXPERIMENT 2: HOUGH TRANSFORM (varying threshold) ===")
    results = []
    
    for i, (threshold, comment) in enumerate(thresholds, 1):
        # Create color copy for drawing
        img_lines = img.copy()
        
        # Hough Lines Probabilistic
        lines = cv2.HoughLinesP(
            edges,
            rho=1,
            theta=np.pi/180,
            threshold=threshold,
            minLineLength=30,
            maxLineGap=10
        )
        
        # Draw lines
        line_count = 0
        if lines is not None:
            for line in lines:
                x1, y1, x2, y2 = line[0]
                cv2.line(img_lines, (x1, y1), (x2, y2), (0, 255, 0), 2)
                line_count += 1
        
        filename = f"hough_{i}_thresh{threshold}"
        save_image(img_lines, filename)
        
        print(f"Hough {i}: Threshold={threshold}, Lines detected={line_count}")
        results.append({
            'parameter': 'threshold',
            'value': threshold,
            'filename': filename,
            'comment': f"{comment} ({line_count} lines)"
        })
    
    return results

# ============= EXPERIMENT 3: HSV vs RGB COLOR SPACES =============

def detect_yellow_rgb(img):
    """Detect yellow lanes using RGB color space"""
    # Yellow in RGB: high Red, high Green, low Blue
    lower_yellow = np.array([100, 100, 0])
    upper_yellow = np.array([255, 255, 100])
    
    mask = cv2.inRange(img, lower_yellow, upper_yellow)
    result = cv2.bitwise_and(img, img, mask=mask)
    
    save_image(result, "yellow_rgb")
    return result, mask

def detect_yellow_hsv(img):
    """Detect yellow lanes using HSV color space"""
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    
    # Yellow in HSV: Hue ~20-30
    lower_yellow = np.array([20, 100, 100])
    upper_yellow = np.array([30, 255, 255])
    
    mask = cv2.inRange(hsv, lower_yellow, upper_yellow)
    result = cv2.bitwise_and(img, img, mask=mask)
    
    save_image(result, "yellow_hsv")
    return result, mask

def experiment_color_spaces(img):
    """Compare HSV vs RGB for yellow lane detection"""
    
    print("\n=== EXPERIMENT 3: HSV vs RGB COLOR SPACES ===")
    
    print("RGB detection...")
    rgb_result, rgb_mask = detect_yellow_rgb(img)
    rgb_pixels = np.sum(rgb_mask > 0)
    
    print("HSV detection...")
    hsv_result, hsv_mask = detect_yellow_hsv(img)
    hsv_pixels = np.sum(hsv_mask > 0)
    
    print(f"RGB: {rgb_pixels} yellow pixels detected")
    print(f"HSV: {hsv_pixels} yellow pixels detected")
    
    return {
        'rgb': {'result': rgb_result, 'pixels': rgb_pixels},
        'hsv': {'result': hsv_result, 'pixels': hsv_pixels}
    }

# ============= EXPERIMENT 4: LIGHTING CONDITIONS =============

def adjust_brightness(img, factor):
    """Adjust image brightness (factor > 1 = brighter, < 1 = darker)"""
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    h, s, v = cv2.split(hsv)
    
    v = np.clip(v * factor, 0, 255).astype(np.uint8)
    
    hsv_adjusted = cv2.merge([h, s, v])
    return cv2.cvtColor(hsv_adjusted, cv2.COLOR_HSV2BGR)

def experiment_lighting(img):
    """Test lane detection under different lighting conditions"""
    
    print("\n=== EXPERIMENT 4: LIGHTING CONDITIONS ===")
    
    # Create bright and dark versions
    dark_img = adjust_brightness(img, 0.5)
    bright_img = adjust_brightness(img, 1.5)
    
    save_image(dark_img, "lighting_dark")
    save_image(bright_img, "lighting_bright")
    
    # Use same HSV parameters for both
    def detect_lines(test_img, name):
        hsv = cv2.cvtColor(test_img, cv2.COLOR_BGR2HSV)
        lower_yellow = np.array([20, 100, 100])
        upper_yellow = np.array([30, 255, 255])
        mask = cv2.inRange(hsv, lower_yellow, upper_yellow)
        
        # Apply edge detection and Hough
        gray = cv2.cvtColor(test_img, cv2.COLOR_BGR2GRAY)
        edges = cv2.Canny(gray, 100, 200)
        lines = cv2.HoughLinesP(edges, 1, np.pi/180, 50, 
                                minLineLength=30, maxLineGap=10)
        
        result = test_img.copy()
        line_count = 0
        if lines is not None:
            for line in lines:
                x1, y1, x2, y2 = line[0]
                cv2.line(result, (x1, y1), (x2, y2), (0, 255, 0), 2)
                line_count += 1
        
        save_image(result, f"lighting_{name}_detected")
        return line_count
    
    print("Testing normal lighting...")
    normal_lines = detect_lines(img, "normal")
    
    print("Testing dark lighting...")
    dark_lines = detect_lines(dark_img, "dark")
    
    print("Testing bright lighting...")
    bright_lines = detect_lines(bright_img, "bright")
    
    print(f"Lines detected - Normal: {normal_lines}, Dark: {dark_lines}, Bright: {bright_lines}")
    
    return {
        'normal': normal_lines,
        'dark': dark_lines,
        'bright': bright_lines
    }

# ============= MAIN EXECUTION =============

def main():
    print("Loading image...")
    img = load_image()
    print(f"Image loaded: {img.shape}")
    
    # Run all experiments
    canny_results = experiment_canny(img)
    hough_results = experiment_hough(img, None)
    color_results = experiment_color_spaces(img)
    lighting_results = experiment_lighting(img)
    
    print("\n=== ALL EXPERIMENTS COMPLETE ===")
    print(f"Results saved to: {OUTPUT_DIR}")
    print("\nNext steps:")
    print("1. Review images in outputs/experiments/")
    print("2. Fill in tables in report template")
    print("3. Add screenshots to report")
    print("4. Write critical reflection")

if __name__ == '__main__':
    main()
EOFPYTHON

#!/usr/bin/env python3

import rosbag
import cv2
from cv_bridge import CvBridge
import sys

bag_file = sys.argv[1]
output_video = "/mnt/user-data/outputs/bag_video.mp4"

bridge = CvBridge()
fourcc = cv2.VideoWriter_fourcc(*'mp4v')
writer = None

print(f"Converting {bag_file} to video...")

bag = rosbag.Bag(bag_file)
frame_count = 0

for topic, msg, t in bag.read_messages(topics=['/deakinbot/camera_node/image/compressed']):
    cv_image = bridge.compressed_imgmsg_to_cv2(msg, "bgr8")
    
    if writer is None:
        h, w = cv_image.shape[:2]
        writer = cv2.VideoWriter(output_video, fourcc, 20.0, (w, h))
    
    writer.write(cv_image)
    frame_count += 1
    
    if frame_count % 10 == 0:
        print(f"Processed {frame_count} frames...")

bag.close()
writer.release()
print(f"Video saved: {output_video} ({frame_count} frames)")
EOF
